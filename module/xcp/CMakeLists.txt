# XCP 主机协议核心（A 块）+ 单测。MSVC: cmake -S . -B build && cmake --build build --config Release && ctest -C Release
cmake_minimum_required(VERSION 3.10)
project(xcp_core C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_options(/utf-8 /W4)
  add_compile_definitions(_CRT_SECURE_NO_WARNINGS _WINSOCK_DEPRECATED_NO_WARNINGS)
else()
  add_compile_options(-Wall -Wextra -Wvla)
  if(NOT WIN32)
    add_compile_definitions(_POSIX_C_SOURCE=200809L)
  endif()
endif()
add_library(xcp_core STATIC xcp_plat.c xcp_eth.c xcp_session.c xcp_checksum.c)
target_include_directories(xcp_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(WIN32)
  target_link_libraries(xcp_core PUBLIC ws2_32)
else()
  find_package(Threads REQUIRED)
  target_link_libraries(xcp_core PUBLIC Threads::Threads)
endif()
enable_testing()
add_executable(xcp_session_test tests/xcp_session_test.c tests/xcp_fake_slave.c)
target_link_libraries(xcp_session_test xcp_core)
if(WIN32)
  target_link_libraries(xcp_session_test winmm)
endif()
add_test(NAME xcp_session_test COMMAND xcp_session_test)
set_tests_properties(xcp_session_test PROPERTIES TIMEOUT 300)
